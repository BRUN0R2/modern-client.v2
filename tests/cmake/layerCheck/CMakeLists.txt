# Empty modules that run the layer check (cmake/nxLayers.cmake) on one case, chosen with
# -DNX_LAYER_CASE=<case>: the regression tests of the rule in docs/01-architecture.md, section 1.1.
cmake_minimum_required(VERSION 4.3)
project(nxLayerCheckFixture LANGUAGES NONE)

list(APPEND CMAKE_MODULE_PATH "${NX_SOURCE_DIR}/cmake")
include(nxLayers)

add_library(thirdParty INTERFACE) # declares no layer

# Writes the module unit `name`, of the text `content`, in the binary directory, and its path in `out`.
function(fixtureUnit name content out)
    set(path "${CMAKE_CURRENT_BINARY_DIR}/${name}")
    file(WRITE "${path}" "${content}")
    set(${out} "${path}" PARENT_SCOPE)
endfunction()

if(NX_LAYER_CASE STREQUAL "lowerLayers")
    add_library(base INTERFACE)
    nxDeclareModuleLayer(base 0 DEPENDS thirdParty)
    add_library(middle INTERFACE)
    add_library(fixture::middle ALIAS middle)
    nxDeclareModuleLayer(middle 1 DEPENDS base)
    add_library(top INTERFACE)
    nxDeclareModuleLayer(top 2 DEPENDS fixture::middle base "$<$<BOOL:ON>:thirdParty>")
elseif(NX_LAYER_CASE STREQUAL "higherLayerDefinedLater")
    # The dependency does not exist yet when `base` is declared.
    add_library(base INTERFACE)
    nxDeclareModuleLayer(base 0 DEPENDS top)
    add_library(top INTERFACE)
    nxDeclareModuleLayer(top 2)
elseif(NX_LAYER_CASE STREQUAL "sameLayer")
    add_library(left INTERFACE)
    nxDeclareModuleLayer(left 1)
    add_library(right INTERFACE)
    nxDeclareModuleLayer(right 1 DEPENDS left)
elseif(NX_LAYER_CASE STREQUAL "declaredLateral")
    # A backend on the interface it implements, declared in the catalog.
    add_library(interface INTERFACE)
    add_library(fixture::interface ALIAS interface)
    nxDeclareModuleLayer(interface 1)
    add_library(backend INTERFACE)
    nxDeclareModuleLayer(backend 1 DEPENDS fixture::interface LATERAL fixture::interface)
elseif(NX_LAYER_CASE STREQUAL "lateralOfAnotherLayer")
    add_library(base INTERFACE)
    nxDeclareModuleLayer(base 0)
    add_library(top INTERFACE)
    nxDeclareModuleLayer(top 1 DEPENDS base LATERAL base)
elseif(NX_LAYER_CASE STREQUAL "layerZero")
    add_library(base INTERFACE)
    nxDeclareModuleLayer(base 0)
    add_library(other INTERFACE)
    nxDeclareModuleLayer(other 0 DEPENDS base)
elseif(NX_LAYER_CASE MATCHES "^(declaredImports|undeclaredImport|dependencyNotImported)$")
    fixtureUnit(base.cppm "export module fixture.base;\n" baseUnit)
    add_library(base INTERFACE)
    nxDeclareModuleLayer(base 0 SOURCES "${baseUnit}")
    fixtureUnit(middle.cppm "export module fixture.middle;\nimport std;\nimport fixture.base;\n" middleUnit)
    add_library(middle INTERFACE)
    add_library(fixture::middle ALIAS middle)
    nxDeclareModuleLayer(middle 1 DEPENDS base SOURCES "${middleUnit}")
    if(NX_LAYER_CASE STREQUAL "declaredImports")
        # Imported by its implementation unit, through its alias; comments and partitions import nothing.
        set(topInterface "export module fixture.top;\nexport import :part;\n// import fixture.base;\n")
        set(topImplementation "module fixture.top;\nimport fixture.middle;\n")
        set(topDepends thirdParty fixture::middle)
    elseif(NX_LAYER_CASE STREQUAL "undeclaredImport")
        # Linked through `middle`, it builds.
        set(topInterface "export module fixture.top;\nimport fixture.middle;\n")
        set(topImplementation "module fixture.top;\nimport fixture.base;\n")
        set(topDepends fixture::middle)
    else()
        set(topInterface "export module fixture.top;\n")
        set(topImplementation "module fixture.top;\nimport fixture.middle;\n")
        set(topDepends fixture::middle base)
    endif()
    fixtureUnit(top.cppm "${topInterface}" topInterfaceUnit)
    fixtureUnit(top.cpp "${topImplementation}" topImplementationUnit)
    add_library(top INTERFACE)
    nxDeclareModuleLayer(top 2 DEPENDS ${topDepends} SOURCES "${topInterfaceUnit}" "${topImplementationUnit}")
elseif(NX_LAYER_CASE MATCHES "^independentTree")
    # A directory like rook/, whose CMakeLists.txt this fixture writes, beside a module of the project.
    add_library(importedLibrary INTERFACE IMPORTED GLOBAL)
    add_library(allowedStd INTERFACE)
    add_library(fixture::std ALIAS allowedStd)
    fixtureUnit(engine.cppm "export module fixture.engine;\n" engineUnit)
    add_library(engine INTERFACE)
    add_library(fixture::engine ALIAS engine)
    nxDeclareModuleLayer(engine 0 SOURCES "${engineUnit}")
    # Not declared to anything, as a target of rook/ is not: the check finds it by walking its directory.
    add_library(helper INTERFACE)

    set(treeUnit "export module fixture.tree;\nimport std;\n// import fixture.engine;\n")
    set(treeLinks "fixture::std importedLibrary inner")
    if(NX_LAYER_CASE STREQUAL "independentTreeLinksOutside")
        set(treeLinks "fixture::std helper")
    elseif(NX_LAYER_CASE STREQUAL "independentTreeLinksModule")
        set(treeLinks "fixture::engine")
    elseif(NX_LAYER_CASE STREQUAL "independentTreeLinksInExpression")
        set(treeLinks "$<$<BOOL:ON>:fixture::engine>")
    elseif(NX_LAYER_CASE STREQUAL "independentTreeImportsModule")
        set(treeUnit "export module fixture.tree;\nimport fixture.engine;\n")
    endif()
    set(treeDirectory "${CMAKE_CURRENT_BINARY_DIR}/tree/source")
    file(WRITE "${treeDirectory}/tree.cppm" "${treeUnit}")
    file(WRITE "${treeDirectory}/CMakeLists.txt"
        "add_library(inner INTERFACE)\n"
        "add_library(tree INTERFACE tree.cppm)\n"
        "target_link_libraries(tree INTERFACE ${treeLinks})\n")
    add_subdirectory("${treeDirectory}" tree/binary)
    nxDeclareIndependentTree("${treeDirectory}" ALLOW fixture::std)

    # The other way round is free: a module of the project may depend on what stands apart.
    add_library(user INTERFACE)
    nxDeclareModuleLayer(user 1 DEPENDS tree)
else()
    message(FATAL_ERROR "layerCheck: unknown case '${NX_LAYER_CASE}'")
endif()
